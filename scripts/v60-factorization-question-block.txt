  const FACTORIZATION_TYPES = Object.freeze([
    "common-factor",
    "grouping",
    "difference-squares",
    "perfect-square-trinomial",
    "general-trinomial"
  ]);
  const FACTORIZATION_CASE_LABELS = Object.freeze({
    "common-factor": "CASO 1 · FACTOR COMÚN",
    grouping: "CASO 2 · AGRUPACIÓN",
    "difference-squares": "CASO 3 · DIFERENCIA DE CUADRADOS",
    "perfect-square-trinomial": "CASO 4 · TRINOMIO CUADRADO PERFECTO",
    "general-trinomial": "CASO 5 · TRINOMIO GENERAL"
  });

  function choice(items) { return items[Math.floor(Math.random() * items.length)]; }
  function shuffled(items) {
    const result = [...items];
    for (let index = result.length - 1; index > 0; index--) {
      const swap = Math.floor(Math.random() * (index + 1));
      [result[index], result[swap]] = [result[swap], result[index]];
    }
    return result;
  }
  function randomInt(min, max) { return Math.floor(Math.random() * (max - min + 1)) + min; }
  function randomSigned(min = 1, max = 8) { return randomInt(min, max) * (Math.random() < 0.5 ? -1 : 1); }
  function gcdInt(a, b) {
    let x = Math.abs(Math.trunc(a));
    let y = Math.abs(Math.trunc(b));
    while (y) [x, y] = [y, x % y];
    return x;
  }
  function coprime(a, b) { return gcdInt(a, b) === 1; }
  function rememberQuestion(question) {
    if (state.questionKeys.includes(question.key)) return false;
    state.questionKeys.push(question.key);
    if (state.questionKeys.length > QUESTION_MEMORY_LIMIT) state.questionKeys.shift();
    return true;
  }
  function variablePart(power) {
    if (power === 0) return "";
    if (power === 1) return "x";
    if (power === 2) return "x²";
    return `x^${power}`;
  }
  function formatPolynomial(terms) {
    let output = "";
    for (const { coef, power } of terms) {
      if (!coef) continue;
      const negative = coef < 0;
      const absolute = Math.abs(coef);
      const variable = variablePart(power);
      const coefficient = variable && absolute === 1 ? "" : String(absolute);
      const body = `${coefficient}${variable}` || "0";
      if (!output) output = negative ? `−${body}` : body;
      else output += negative ? ` − ${body}` : ` + ${body}`;
    }
    return output || "0";
  }
  function formatLinear(a, b) {
    return formatPolynomial([{ coef: a, power: 1 }, { coef: b, power: 0 }]);
  }
  function factor(a, b) { return `(${formatLinear(a, b)})`; }
  function factoredPair(a, b, c, d) { return `${factor(a, b)}${factor(c, d)}`; }
  function polynomialSignature(c2 = 0, c1 = 0, c0 = 0) {
    return `${Number(c2)}|${Number(c1)}|${Number(c0)}`;
  }
  function pairCoefficients(a, b, c, d) { return [a * c, a * d + b * c, b * d]; }
  function pairSignature(a, b, c, d) { return polynomialSignature(...pairCoefficients(a, b, c, d)); }
  function optionEntry(text, signature) { return { text, signature }; }
  function makeVerifiedOptions(correctText, correctSignature, candidates) {
    const entries = [optionEntry(correctText, correctSignature)];
    const seenText = new Set([correctText]);
    const seenSignature = new Set([correctSignature]);
    for (const candidate of candidates.filter(Boolean)) {
      if (!candidate.text || !candidate.signature) continue;
      if (seenText.has(candidate.text) || seenSignature.has(candidate.signature)) continue;
      seenText.add(candidate.text);
      seenSignature.add(candidate.signature);
      entries.push(candidate);
      if (entries.length >= 4) break;
    }
    let guard = 0;
    while (entries.length < 4 && guard++ < 120) {
      const a = randomInt(1, 4), c = randomInt(1, 4);
      const b = randomSigned(1, 7), d = randomSigned(1, 7);
      const text = factoredPair(a, b, c, d);
      const signature = pairSignature(a, b, c, d);
      if (seenText.has(text) || seenSignature.has(signature)) continue;
      seenText.add(text); seenSignature.add(signature);
      entries.push(optionEntry(text, signature));
    }
    if (entries.length < 4) throw new Error("Could not build four algebraically distinct factorization options.");
    return shuffled(entries).map((entry) => entry.text);
  }

  function createCommonFactorQuestion() {
    const g = randomInt(2, 9);
    let a, b;
    do {
      a = randomInt(1, 7);
      b = randomSigned(1, 9);
    } while (!coprime(a, b));
    const coefficients = [g * a, g * b, 0];
    const display = formatPolynomial([{ coef: coefficients[0], power: 2 }, { coef: coefficients[1], power: 1 }]);
    const correct = `${g}x${factor(a, b)}`;
    const correctSignature = polynomialSignature(...coefficients);
    return {
      type: "common-factor",
      caseLabel: FACTORIZATION_CASE_LABELS["common-factor"],
      display,
      coefficients,
      correctSignature,
      rule: "Extrae el MCD y la menor potencia común de x.",
      steps: ["Calcula el máximo factor numérico común.", "Extrae también x porque aparece en todos los términos."],
      correct,
      explanation: `El MCD de los coeficientes es ${g} y la menor potencia común es x; por eso el factor común completo es ${g}x y queda ${formatLinear(a, b)}.`,
      options: makeVerifiedOptions(correct, correctSignature, [
        optionEntry(`${g}${factor(a, b)}`, polynomialSignature(0, g * a, g * b)),
        optionEntry(`${g}x${factor(a + 1, b)}`, polynomialSignature(g * (a + 1), g * b, 0)),
        optionEntry(`${g}x${factor(a, -b)}`, polynomialSignature(g * a, -g * b, 0)),
        optionEntry(`${g + 1}x${factor(a, b)}`, polynomialSignature((g + 1) * a, (g + 1) * b, 0))
      ])
    };
  }

  function createGroupingQuestion() {
    let a, b, c, d, middle;
    do {
      a = randomInt(1, 4);
      c = randomInt(1, 4);
      b = randomSigned(1, 7);
      d = randomSigned(1, 7);
      middle = a * d + b * c;
    } while (!coprime(a, b) || !coprime(c, d) || middle === 0 || (a === c && b === d) || (a === c && b === -d));
    const coefficients = pairCoefficients(a, b, c, d);
    const display = formatPolynomial([
      { coef: a * c, power: 2 },
      { coef: a * d, power: 1 },
      { coef: b * c, power: 1 },
      { coef: b * d, power: 0 }
    ]);
    const correct = factoredPair(a, b, c, d);
    const correctSignature = polynomialSignature(...coefficients);
    return {
      type: "grouping",
      caseLabel: FACTORIZATION_CASE_LABELS.grouping,
      display,
      coefficients,
      correctSignature,
      rule: "Agrupa los cuatro términos en dos pares con un binomio común.",
      steps: ["Factoriza el primer par.", "Factoriza el segundo par y verifica el mismo binomio."],
      correct,
      explanation: `La expansión inversa de ${factor(a, b)}${factor(c, d)} produce exactamente los cuatro términos mostrados; ambos binomios son primitivos, así que la factorización ya está completa.`,
      options: makeVerifiedOptions(correct, correctSignature, [
        optionEntry(factoredPair(a, -b, c, d), pairSignature(a, -b, c, d)),
        optionEntry(factoredPair(a, b, c, -d), pairSignature(a, b, c, -d)),
        optionEntry(factoredPair(a, d, c, b), pairSignature(a, d, c, b)),
        optionEntry(`${factor(a * c, b)}${factor(1, d)}`, pairSignature(a * c, b, 1, d))
      ])
    };
  }

  function createDifferenceSquaresQuestion() {
    let a, b;
    do {
      a = randomInt(1, 7);
      b = randomInt(2, 12);
    } while (!coprime(a, b));
    const coefficients = [a * a, 0, -(b * b)];
    const display = formatPolynomial([{ coef: coefficients[0], power: 2 }, { coef: coefficients[2], power: 0 }]);
    const correct = `${factor(a, -b)}${factor(a, b)}`;
    const correctSignature = polynomialSignature(...coefficients);
    return {
      type: "difference-squares",
      caseLabel: FACTORIZATION_CASE_LABELS["difference-squares"],
      display,
      coefficients,
      correctSignature,
      rule: "A² − B² = (A − B)(A + B).",
      steps: ["Comprueba que ambos términos sean cuadrados perfectos.", "Usa binomios conjugados y verifica que no quede factor común."],
      correct,
      explanation: `${a * a}x² = (${a}x)² y ${b * b} = ${b}². Como ${a} y ${b} son coprimos, no queda un factor numérico común adicional.`,
      options: makeVerifiedOptions(correct, correctSignature, [
        optionEntry(`${factor(a, -b)}²`, pairSignature(a, -b, a, -b)),
        optionEntry(`${factor(a, b)}²`, pairSignature(a, b, a, b)),
        optionEntry(`${factor(a * a, -b)}${factor(1, b)}`, pairSignature(a * a, -b, 1, b)),
        optionEntry(`${factor(a, -(b * b))}${factor(a, 1)}`, pairSignature(a, -(b * b), a, 1))
      ])
    };
  }

  function createPerfectSquareTrinomialQuestion() {
    let a, b;
    do {
      a = randomInt(1, 6);
      b = randomInt(1, 9);
    } while (!coprime(a, b));
    const sign = Math.random() < 0.5 ? -1 : 1;
    const signedB = sign * b;
    const coefficients = [a * a, 2 * a * signedB, b * b];
    const display = formatPolynomial([
      { coef: coefficients[0], power: 2 },
      { coef: coefficients[1], power: 1 },
      { coef: coefficients[2], power: 0 }
    ]);
    const correct = `${factor(a, signedB)}²`;
    const correctSignature = polynomialSignature(...coefficients);
    return {
      type: "perfect-square-trinomial",
      caseLabel: FACTORIZATION_CASE_LABELS["perfect-square-trinomial"],
      display,
      coefficients,
      correctSignature,
      rule: "A² ± 2AB + B² = (A ± B)².",
      steps: ["Verifica los cuadrados perfectos de los extremos.", "Confirma que el término medio sea ±2AB y que el binomio sea primitivo."],
      correct,
      explanation: `Los extremos son (${a}x)² y ${b}², el término medio es ${sign < 0 ? "−" : "+"}2(${a}x)(${b}) y MCD(${a}, ${b}) = 1.`,
      options: makeVerifiedOptions(correct, correctSignature, [
        optionEntry(`${factor(a, -signedB)}²`, pairSignature(a, -signedB, a, -signedB)),
        optionEntry(`${factor(a, -b)}${factor(a, b)}`, pairSignature(a, -b, a, b)),
        optionEntry(`${factor(a * a, signedB)}²`, pairSignature(a * a, signedB, a * a, signedB)),
        optionEntry(factoredPair(a, signedB, 1, signedB), pairSignature(a, signedB, 1, signedB))
      ])
    };
  }

  function createGeneralTrinomialQuestion() {
    let a, b, c, d, middle;
    do {
      a = randomInt(1, 3);
      c = randomInt(1, 3);
      b = randomSigned(1, 8);
      d = randomSigned(1, 8);
      middle = a * d + b * c;
    } while (!coprime(a, b) || !coprime(c, d) || (a === c && b === d) || middle === 0);
    const coefficients = pairCoefficients(a, b, c, d);
    const display = formatPolynomial([
      { coef: coefficients[0], power: 2 },
      { coef: coefficients[1], power: 1 },
      { coef: coefficients[2], power: 0 }
    ]);
    const correct = factoredPair(a, b, c, d);
    const correctSignature = polynomialSignature(...coefficients);
    return {
      type: "general-trinomial",
      caseLabel: FACTORIZATION_CASE_LABELS["general-trinomial"],
      display,
      coefficients,
      correctSignature,
      rule: "Reconstruye dos binomios primitivos cuyo producto produzca Ax² + Bx + C.",
      steps: ["Relaciona el producto de los primeros términos con el coeficiente de x².", "Verifica término medio y término independiente al multiplicar."],
      correct,
      explanation: `Al multiplicar ${factor(a, b)} por ${factor(c, d)} se recuperan exactamente ${coefficients[0]}x², ${coefficients[1]}x y ${coefficients[2]}; además, cada binomio es primitivo.`,
      options: makeVerifiedOptions(correct, correctSignature, [
        optionEntry(factoredPair(a, -b, c, d), pairSignature(a, -b, c, d)),
        optionEntry(factoredPair(a, b, c, -d), pairSignature(a, b, c, -d)),
        optionEntry(factoredPair(a, d, c, b), pairSignature(a, d, c, b)),
        optionEntry(factoredPair(c, b, a, d), pairSignature(c, b, a, d))
      ])
    };
  }

  function generateQuestionCandidate(type) {
    const builders = {
      "common-factor": createCommonFactorQuestion,
      grouping: createGroupingQuestion,
      "difference-squares": createDifferenceSquaresQuestion,
      "perfect-square-trinomial": createPerfectSquareTrinomialQuestion,
      "general-trinomial": createGeneralTrinomialQuestion
    };
    const question = builders[type]();
    question.id = `${type}-${question.display.replace(/\s+/g, "-")}`;
    question.key = `${type}|${question.display}|${question.correct}`;
    question.prompt = `Factoriza completamente: ${question.display}`;
    return question;
  }

  function createQuestion() {
    const types = shuffled(FACTORIZATION_TYPES);
    for (let attempt = 0; attempt < 180; attempt++) {
      const candidate = generateQuestionCandidate(types[attempt % types.length]);
      if (rememberQuestion(candidate)) return candidate;
    }
    state.questionKeys.length = 0;
    const fallback = generateQuestionCandidate(choice(FACTORIZATION_TYPES));
    rememberQuestion(fallback);
    return fallback;
  }

  function drawWrappedQuestionText(context, text, x, y, maxWidth, lineHeight, maxLines = 4) {
    const words = String(text).split(/\s+/);
    let line = "";
    let lineIndex = 0;
    for (const word of words) {
      const test = line ? `${line} ${word}` : word;
      if (context.measureText(test).width > maxWidth && line) {
        context.fillText(line, x, y + lineIndex * lineHeight);
        lineIndex++;
        line = word;
        if (lineIndex >= maxLines - 1) break;
      } else line = test;
    }
    if (lineIndex < maxLines) context.fillText(line, x, y + lineIndex * lineHeight);
  }
