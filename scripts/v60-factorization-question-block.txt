  const FACTORIZATION_TYPES = Object.freeze([
    "common-difference-squares",
    "common-perfect-square",
    "common-general-trinomial",
    "grouping-difference-squares",
    "common-grouping"
  ]);
  const FACTORIZATION_CASE_LABELS = Object.freeze({
    "common-difference-squares": "COMBO 1 · FACTOR COMÚN → DIFERENCIA DE CUADRADOS",
    "common-perfect-square": "COMBO 2 · FACTOR COMÚN → TRINOMIO CUADRADO PERFECTO",
    "common-general-trinomial": "COMBO 3 · FACTOR COMÚN → TRINOMIO GENERAL",
    "grouping-difference-squares": "COMBO 4 · AGRUPACIÓN → DIFERENCIA DE CUADRADOS",
    "common-grouping": "COMBO 5 · FACTOR COMÚN → AGRUPACIÓN"
  });

  function choice(items) { return items[Math.floor(Math.random() * items.length)]; }
  function shuffled(items) {
    const result = [...items];
    for (let index = result.length - 1; index > 0; index--) {
      const swap = Math.floor(Math.random() * (index + 1));
      [result[index], result[swap]] = [result[swap], result[index]];
    }
    return result;
  }
  function randomInt(min, max) { return Math.floor(Math.random() * (max - min + 1)) + min; }
  function randomSigned(min = 1, max = 8) { return randomInt(min, max) * (Math.random() < 0.5 ? -1 : 1); }
  function gcdInt(a, b) {
    let x = Math.abs(Math.trunc(a));
    let y = Math.abs(Math.trunc(b));
    while (y) [x, y] = [y, x % y];
    return x;
  }
  function coprime(a, b) { return gcdInt(a, b) === 1; }
  function rememberQuestion(question) {
    if (state.questionKeys.includes(question.key)) return false;
    state.questionKeys.push(question.key);
    if (state.questionKeys.length > QUESTION_MEMORY_LIMIT) state.questionKeys.shift();
    return true;
  }
  function normalizePoly(coefficients) {
    const result = [...coefficients].map(Number);
    while (result.length > 1 && result[0] === 0) result.shift();
    return result;
  }
  function polySignature(coefficients) { return normalizePoly(coefficients).join("|"); }
  function polyAdd(left, right) {
    const size = Math.max(left.length, right.length);
    const a = Array(size - left.length).fill(0).concat(left);
    const b = Array(size - right.length).fill(0).concat(right);
    return normalizePoly(a.map((value, index) => value + b[index]));
  }
  function polyScale(coefficients, scalar) { return normalizePoly(coefficients.map((value) => value * scalar)); }
  function polyMultiply(left, right) {
    const result = Array(left.length + right.length - 1).fill(0);
    for (let i = 0; i < left.length; i++) for (let j = 0; j < right.length; j++) result[i + j] += left[i] * right[j];
    return normalizePoly(result);
  }
  function linearPoly(a, b) { return [a, b]; }
  function monomialPoly(coef, power) { return [coef, ...Array(power).fill(0)]; }
  function pairPoly(a, b, c, d) { return polyMultiply(linearPoly(a, b), linearPoly(c, d)); }

  function variablePart(power) {
    if (power === 0) return "";
    if (power === 1) return "x";
    if (power === 2) return "x²";
    if (power === 3) return "x³";
    if (power === 4) return "x⁴";
    return `x^${power}`;
  }
  function formatPolynomialFromCoefficients(coefficients) {
    const values = normalizePoly(coefficients);
    const degree = values.length - 1;
    let output = "";
    values.forEach((coef, index) => {
      if (!coef) return;
      const power = degree - index;
      const negative = coef < 0;
      const absolute = Math.abs(coef);
      const variable = variablePart(power);
      const coefficient = variable && absolute === 1 ? "" : String(absolute);
      const body = `${coefficient}${variable}` || "0";
      if (!output) output = negative ? `−${body}` : body;
      else output += negative ? ` − ${body}` : ` + ${body}`;
    });
    return output || "0";
  }
  function formatLinear(a, b) { return formatPolynomialFromCoefficients([a, b]); }
  function formatQuadraticNoLinear(a, b) { return formatPolynomialFromCoefficients([a, 0, b]); }
  function factor(a, b) { return `(${formatLinear(a, b)})`; }
  function quadraticFactor(a, b) { return `(${formatQuadraticNoLinear(a, b)})`; }
  function monomialText(coef, power) {
    const coefficient = coef === 1 && power > 0 ? "" : String(coef);
    return `${coefficient}${variablePart(power)}` || "1";
  }
  function normalizedVisibleText(text) { return String(text).replaceAll("−", "-").replace(/\s+/g, ""); }

  function optionEntry(text, coefficients, isCorrect = false) {
    return { text, coefficients: normalizePoly(coefficients), signature: polySignature(coefficients), isCorrect };
  }
  function genericDistractor(degree) {
    if (degree >= 4) {
      const a = randomInt(1, 4), b = randomInt(1, 8), c = randomInt(1, 4), d = randomSigned(1, 7);
      const coefficients = polyMultiply([1, 0], polyMultiply([a, 0, b], [c, d]));
      return optionEntry(`x${quadraticFactor(a, b)}${factor(c, d)}`, coefficients);
    }
    const g = randomInt(1, 7), a = randomInt(1, 4), c = randomInt(1, 4), b = randomSigned(1, 7), d = randomSigned(1, 7);
    const coefficients = polyMultiply(monomialPoly(g, 1), pairPoly(a, b, c, d));
    return optionEntry(`${monomialText(g, 1)}${factor(a, b)}${factor(c, d)}`, coefficients);
  }
  function makeVerifiedOptions(correctEntry, candidates) {
    const entries = [];
    const seenText = new Set();
    const seenSignature = new Set();
    function add(entry) {
      if (!entry || !entry.text || !entry.signature) return false;
      const textKey = normalizedVisibleText(entry.text);
      if (seenText.has(textKey) || seenSignature.has(entry.signature)) return false;
      seenText.add(textKey); seenSignature.add(entry.signature); entries.push(entry); return true;
    }
    if (!add(correctEntry)) throw new Error("Correct option could not be registered.");
    for (const candidate of candidates) {
      if (entries.length >= 4) break;
      add(candidate);
    }
    let guard = 0;
    const degree = correctEntry.coefficients.length - 1;
    while (entries.length < 4 && guard++ < 300) add(genericDistractor(degree));
    if (entries.length !== 4) throw new Error("Could not build exactly four distinct answer options.");
    const result = shuffled(entries).map((entry, index) => ({ ...entry, id: `option-${index + 1}` }));
    const correct = result.filter((entry) => entry.isCorrect);
    if (correct.length !== 1) throw new Error("Question must contain exactly one correct option.");
    if (new Set(result.map((entry) => normalizedVisibleText(entry.text))).size !== 4) throw new Error("Duplicate visible option text detected.");
    if (new Set(result.map((entry) => entry.signature)).size !== 4) throw new Error("Algebraically duplicate option detected.");
    return result;
  }
  function finalizeQuestion(base, correctEntry, distractors) {
    const options = makeVerifiedOptions({ ...correctEntry, isCorrect: true }, distractors.map((entry) => ({ ...entry, isCorrect: false })));
    const correctOption = options.find((entry) => entry.isCorrect);
    return {
      ...base,
      coefficients: normalizePoly(correctEntry.coefficients),
      correctSignature: correctEntry.signature,
      correct: correctOption.text,
      correctOptionId: correctOption.id,
      options
    };
  }

  function createCommonDifferenceSquaresQuestion() {
    const g = randomInt(2, 7);
    let a, b;
    do { a = randomInt(1, 6); b = randomInt(2, 10); } while (!coprime(a, b));
    const inner = polyMultiply([a, -b], [a, b]);
    const coefficients = polyMultiply(monomialPoly(g, 1), inner);
    const correctText = `${monomialText(g, 1)}${factor(a, -b)}${factor(a, b)}`;
    return finalizeQuestion({
      type: "common-difference-squares",
      caseLabel: FACTORIZATION_CASE_LABELS["common-difference-squares"],
      display: formatPolynomialFromCoefficients(coefficients),
      rule: "No te detengas en el primer factor: extrae factor común y vuelve a factorizar el binomio restante.",
      steps: ["Extrae el MCD y la menor potencia de x.", "Reconoce A² − B² y usa binomios conjugados."],
      explanation: `Primero sale ${monomialText(g, 1)}. El factor restante es una diferencia de cuadrados: (${a}x)² − ${b}².`
    }, optionEntry(correctText, coefficients), [
      optionEntry(`${g}${factor(a, -b)}${factor(a, b)}`, polyScale(inner, g)),
      optionEntry(`${monomialText(g, 1)}${factor(a, -b)}²`, polyMultiply(monomialPoly(g, 1), polyMultiply([a, -b], [a, -b]))),
      optionEntry(`${monomialText(g, 1)}${factor(a, b)}²`, polyMultiply(monomialPoly(g, 1), polyMultiply([a, b], [a, b]))),
      optionEntry(`${monomialText(g + 1, 1)}${factor(a, -b)}${factor(a, b)}`, polyMultiply(monomialPoly(g + 1, 1), inner)),
      optionEntry(`${monomialText(g, 1)}${factor(a, -(b + 1))}${factor(a, b + 1)}`, polyMultiply(monomialPoly(g, 1), polyMultiply([a, -(b + 1)], [a, b + 1])))
    ]);
  }

  function createCommonPerfectSquareQuestion() {
    const g = randomInt(2, 7);
    let a, b;
    do { a = randomInt(1, 5); b = randomInt(1, 9); } while (!coprime(a, b));
    const sign = Math.random() < 0.5 ? -1 : 1;
    const signedB = sign * b;
    const square = polyMultiply([a, signedB], [a, signedB]);
    const coefficients = polyMultiply(monomialPoly(g, 1), square);
    const correctText = `${monomialText(g, 1)}${factor(a, signedB)}²`;
    return finalizeQuestion({
      type: "common-perfect-square",
      caseLabel: FACTORIZATION_CASE_LABELS["common-perfect-square"],
      display: formatPolynomialFromCoefficients(coefficients),
      rule: "Aplica dos etapas: factor común y luego trinomio cuadrado perfecto.",
      steps: ["Extrae el factor común completo.", "Comprueba A² ± 2AB + B² y escribe (A ± B)²."],
      explanation: `Después de extraer ${monomialText(g, 1)}, los tres términos forman el cuadrado de ${factor(a, signedB)}.`
    }, optionEntry(correctText, coefficients), [
      optionEntry(`${g}${factor(a, signedB)}²`, polyScale(square, g)),
      optionEntry(`${monomialText(g, 1)}${factor(a, -signedB)}²`, polyMultiply(monomialPoly(g, 1), polyMultiply([a, -signedB], [a, -signedB]))),
      optionEntry(`${monomialText(g, 1)}${factor(a, -b)}${factor(a, b)}`, polyMultiply(monomialPoly(g, 1), polyMultiply([a, -b], [a, b]))),
      optionEntry(`${monomialText(g + 1, 1)}${factor(a, signedB)}²`, polyMultiply(monomialPoly(g + 1, 1), square)),
      optionEntry(`${monomialText(g, 1)}${factor(a + 1, signedB)}²`, polyMultiply(monomialPoly(g, 1), polyMultiply([a + 1, signedB], [a + 1, signedB])))
    ]);
  }

  function createCommonGeneralTrinomialQuestion() {
    const g = randomInt(2, 6);
    let a, b, c, d;
    do {
      a = randomInt(1, 3); c = randomInt(1, 3); b = randomSigned(1, 7); d = randomSigned(1, 7);
    } while (!coprime(a, b) || !coprime(c, d) || (a === c && b === d) || (a === c && b === -d) || a * d + b * c === 0);
    const pair = pairPoly(a, b, c, d);
    const coefficients = polyMultiply(monomialPoly(g, 1), pair);
    const correctText = `${monomialText(g, 1)}${factor(a, b)}${factor(c, d)}`;
    return finalizeQuestion({
      type: "common-general-trinomial",
      caseLabel: FACTORIZATION_CASE_LABELS["common-general-trinomial"],
      display: formatPolynomialFromCoefficients(coefficients),
      rule: "Primero simplifica con factor común; después factoriza completamente el trinomio general.",
      steps: ["Extrae el factor común numérico y x.", "Busca dos binomios cuyo producto reconstruya los tres coeficientes restantes."],
      explanation: `Al extraer ${monomialText(g, 1)} queda un trinomio que se descompone como ${factor(a, b)}${factor(c, d)}.`
    }, optionEntry(correctText, coefficients), [
      optionEntry(`${g}${factor(a, b)}${factor(c, d)}`, polyScale(pair, g)),
      optionEntry(`${monomialText(g, 1)}${factor(a, -b)}${factor(c, d)}`, polyMultiply(monomialPoly(g, 1), pairPoly(a, -b, c, d))),
      optionEntry(`${monomialText(g, 1)}${factor(a, b)}${factor(c, -d)}`, polyMultiply(monomialPoly(g, 1), pairPoly(a, b, c, -d))),
      optionEntry(`${monomialText(g + 1, 1)}${factor(a, b)}${factor(c, d)}`, polyMultiply(monomialPoly(g + 1, 1), pair)),
      optionEntry(`${monomialText(g, 1)}${factor(a + 1, b)}${factor(c, d)}`, polyMultiply(monomialPoly(g, 1), pairPoly(a + 1, b, c, d)))
    ]);
  }

  function createGroupingDifferenceSquaresQuestion() {
    let m, n;
    do { m = randomInt(1, 5); n = randomInt(2, 9); } while (!coprime(m, n));
    const c = randomSigned(1, 6);
    const coefficients = polyMultiply([1, c], polyMultiply([m, -n], [m, n]));
    const correctText = `${factor(1, c)}${factor(m, -n)}${factor(m, n)}`;
    return finalizeQuestion({
      type: "grouping-difference-squares",
      caseLabel: FACTORIZATION_CASE_LABELS["grouping-difference-squares"],
      display: formatPolynomialFromCoefficients(coefficients),
      rule: "Busca primero el binomio común por agrupación; luego factoriza la diferencia de cuadrados que aparece.",
      steps: ["Agrupa los términos para obtener un factor binomial común.", "Factoriza el segundo factor como A² − B²."],
      explanation: `La agrupación produce ${factor(1, c)}((` + `${m}x)² − ${n}²), y la diferencia de cuadrados completa la factorización.`
    }, optionEntry(correctText, coefficients), [
      optionEntry(`${factor(1, -c)}${factor(m, -n)}${factor(m, n)}`, polyMultiply([1, -c], polyMultiply([m, -n], [m, n]))),
      optionEntry(`${factor(1, c)}${factor(m, -n)}²`, polyMultiply([1, c], polyMultiply([m, -n], [m, -n]))),
      optionEntry(`${factor(1, c)}${factor(m, n)}²`, polyMultiply([1, c], polyMultiply([m, n], [m, n]))),
      optionEntry(`x${factor(1, c)}${factor(m, -n)}${factor(m, n)}`, polyMultiply([1, 0], coefficients)),
      optionEntry(`${factor(1, c + (c > 0 ? 1 : -1))}${factor(m, -n)}${factor(m, n)}`, polyMultiply([1, c + (c > 0 ? 1 : -1)], polyMultiply([m, -n], [m, n])))
    ]);
  }

  function createCommonGroupingQuestion() {
    let a, b, c, d;
    do {
      a = randomInt(1, 4); b = randomInt(1, 8); c = randomInt(1, 4); d = randomSigned(1, 7);
    } while (!coprime(a, b) || !coprime(c, d));
    const quadratic = [a, 0, b];
    const coefficients = polyMultiply([1, 0], polyMultiply(quadratic, [c, d]));
    const correctText = `x${quadraticFactor(a, b)}${factor(c, d)}`;
    return finalizeQuestion({
      type: "common-grouping",
      caseLabel: FACTORIZATION_CASE_LABELS["common-grouping"],
      display: formatPolynomialFromCoefficients(coefficients),
      rule: "Extrae el factor común y usa agrupación para obtener dos factores adicionales.",
      steps: ["Saca x, que aparece en todos los términos.", "Agrupa el polinomio restante para obtener un factor cuadrático y uno lineal."],
      explanation: `Después de extraer x, la agrupación reconstruye ${quadraticFactor(a, b)}${factor(c, d)}. El factor cuadrático tiene término constante positivo y es primitivo.`
    }, optionEntry(correctText, coefficients), [
      optionEntry(`${quadraticFactor(a, b)}${factor(c, d)}`, polyMultiply(quadratic, [c, d])),
      optionEntry(`x${quadraticFactor(a, -b)}${factor(c, d)}`, polyMultiply([1, 0], polyMultiply([a, 0, -b], [c, d]))),
      optionEntry(`x${quadraticFactor(a, b)}${factor(c, -d)}`, polyMultiply([1, 0], polyMultiply(quadratic, [c, -d]))),
      optionEntry(`x${quadraticFactor(a + 1, b)}${factor(c, d)}`, polyMultiply([1, 0], polyMultiply([a + 1, 0, b], [c, d]))),
      optionEntry(`x${quadraticFactor(a, b + 1)}${factor(c, d)}`, polyMultiply([1, 0], polyMultiply([a, 0, b + 1], [c, d])))
    ]);
  }

  function generateQuestionCandidate(type) {
    const builders = {
      "common-difference-squares": createCommonDifferenceSquaresQuestion,
      "common-perfect-square": createCommonPerfectSquareQuestion,
      "common-general-trinomial": createCommonGeneralTrinomialQuestion,
      "grouping-difference-squares": createGroupingDifferenceSquaresQuestion,
      "common-grouping": createCommonGroupingQuestion
    };
    const question = builders[type]();
    question.id = `${type}-${question.display.replace(/\s+/g, "-")}`;
    question.key = `${type}|${question.display}|${question.correctSignature}`;
    question.prompt = `Factoriza completamente aplicando más de un caso: ${question.display}`;
    question.comboCases = question.caseLabel.split("·").slice(1).join("·").trim();
    return question;
  }

  function createQuestion() {
    const types = shuffled(FACTORIZATION_TYPES);
    for (let attempt = 0; attempt < 200; attempt++) {
      const candidate = generateQuestionCandidate(types[attempt % types.length]);
      if (rememberQuestion(candidate)) return candidate;
    }
    state.questionKeys.length = 0;
    const fallback = generateQuestionCandidate(choice(FACTORIZATION_TYPES));
    rememberQuestion(fallback);
    return fallback;
  }

  function drawWrappedQuestionText(context, text, x, y, maxWidth, lineHeight, maxLines = 4) {
    const words = String(text).split(/\s+/);
    let line = "";
    let lineIndex = 0;
    for (const word of words) {
      const test = line ? `${line} ${word}` : word;
      if (context.measureText(test).width > maxWidth && line) {
        context.fillText(line, x, y + lineIndex * lineHeight);
        lineIndex++;
        line = word;
        if (lineIndex >= maxLines - 1) break;
      } else line = test;
    }
    if (lineIndex < maxLines) context.fillText(line, x, y + lineIndex * lineHeight);
  }
