  function drawFactorizationPanel(question) {
    const x = 790, y = 70, width = 360, height = 480;
    qctx.fillStyle = "#f7f9fc";
    roundedRect(qctx, x, y, width, height, 20); qctx.fill();
    qctx.strokeStyle = "#c7d2e0"; qctx.lineWidth = 3; qctx.stroke();
    qctx.textAlign = "left"; qctx.textBaseline = "alphabetic";
    qctx.fillStyle = "#4d5c72"; qctx.font = "900 18px Segoe UI";
    drawWrappedQuestionText(qctx, question.caseLabel, x + 28, y + 44, width - 56, 23, 2);
    qctx.fillStyle = "#172033"; qctx.font = "900 21px Segoe UI";
    drawWrappedQuestionText(qctx, question.rule, x + 28, y + 112, width - 56, 28, 3);

    qctx.fillStyle = "#fff4d8";
    roundedRect(qctx, x + 22, y + 205, width - 44, 105, 14); qctx.fill();
    qctx.strokeStyle = "#d89000"; qctx.lineWidth = 2; qctx.stroke();
    qctx.fillStyle = "#9b6400"; qctx.font = "900 18px Segoe UI";
    qctx.fillText("AYUDA INICIAL", x + 38, y + 236);
    qctx.fillStyle = "#3e4654"; qctx.font = "800 18px Segoe UI";
    drawWrappedQuestionText(qctx, `Empieza así: ${question.steps[0]}`, x + 38, y + 270, width - 76, 24, 2);

    qctx.fillStyle = "#2457d6"; qctx.font = "900 18px Segoe UI";
    qctx.fillText("SIGUIENTE PISTA", x + 28, y + 350);
    qctx.fillStyle = "#596579"; qctx.font = "700 18px Segoe UI";
    drawWrappedQuestionText(qctx, question.steps[1], x + 28, y + 386, width - 56, 25, 3);

    qctx.fillText("La ayuda orienta el primer paso; no revela la respuesta.", x + 28, y + 456, width - 56);
  }
